\section{Fundamentos Matemáticos y Decisiones de Implementación}

\subsection{Desagregación Temporal de Denton (1971) sin Indicador}
Para construir la ventana ampliada 1999--2003 sin distorsionar la varianza, se aplicó la técnica de Denton en primeras diferencias:

\begin{equation}
\min_{y} \, (D y)^\top (D y) \quad \text{sujeto a} \quad C y = Y_A
\end{equation}

Donde $D$ es la matriz de primeras diferencias de dimensión $(T-1) \times T$, $C$ es la matriz agregadora anual ($M \times T$, con bloques $[1, 1, 1, 1]$), y $Y_A$ es el vector anual auditado del \textit{Compendio Fiscal de Hacienda}. La solución analítica exacta de mínimos cuadrados restringidos es:

\begin{equation}
y = (D^\top D)^{-1} C^\top \left[ C (D^\top D)^{-1} C^\top \right]^{-1} Y_A
\end{equation}

\textbf{Defensa}: Esta solución no introduce estacionalidad ficticia y minimiza la curvatura entre trimestres adyacentes respetando la restricción de balance contable interanual.

\subsection{Consolidación del Balance Cuasifiscal del BCRA}
La deuda pública relevante para la solvencia intertemporal no es únicamente la del Sector Público No Financiero (SPNF), sino la deuda pública consolidada neta:

\begin{equation}
D_t^{\text{cons}} = D_t^{\text{SPNF}} + \text{Pasivos Remunerados del BCRA}_t - \text{Títulos Públicos en Activo del BCRA}_t
\end{equation}

\subsection{El Sistema VECM y el Teorema de Representación de Granger}
Ante variables no estacionarias $I(1)$ cointegradas, un VAR en diferencias ignora el atractor de largo plazo. El VECM resuelve esta omisión:

\begin{equation}
\Delta Y_t = \mu + \alpha \beta^\top Y_{t-1} + \sum_{i=1}^{p-1} \Gamma_i \Delta Y_{t-i} + \Phi \tilde{y}_t + \varepsilon_t
\end{equation}

Donde $\beta^\top Y_{t-1}$ representa el término de corrección de error (ECT). La velocidad de ajuste del superávit primario hacia el equilibrio intertemporal es:
\begin{itemize}
    \item $\alpha_{pb} = 0.0044$ ($p = 0.204$) en la ventana ampliada ($n=108$).
    \item $\alpha_{pb} = 0.0014$ ($p = 0.559$) en la ventana homogénea ($n=87$).
\end{itemize}

\subsection{Quiebres Estructurales Múltiples de Bai y Perron (2003)}
El algoritmo de partición óptima resuelve por programación dinámica:

\begin{equation}
(\hat{T}_1, \dots, \hat{T}_m) = \arg\min_{(T_1, \dots, T_m)} \sum_{j=1}^{m+1} \sum_{t=T_{j-1}+1}^{T_j} \left( y_t - z_t^\top \delta_j \right)^2
\end{equation}

\subsection{Test de Umbral de Hansen (1999, 2000) y Fatiga Fiscal}
La estimación del modelo de umbral en primera diferencia $\Delta pb_t$ (estacionaria) arrojó:

\begin{equation}
\Delta pb_t = \beta_1 d_{t-1} \cdot \mathbb{I}(\text{EMBI}_{t-1} \le \tau^*) + \beta_2 d_{t-1} \cdot \mathbb{I}(\text{EMBI}_{t-1} > \tau^*) + \gamma X_t + \varepsilon_t
\end{equation}

\section{Análisis de Sostenibilidad de Deuda (DSA) Estocástico}

La ecuación dinámica no lineal de acumulación de deuda sobre PIB se expresa como:

\begin{equation}
d_t = d_{t-1} \left( \frac{1 + r_t}{1 + g_t} \right) - pb_t + sft_t
\end{equation}

Donde la tasa efectiva real pondera moneda doméstica y extranjera:

\begin{equation}
r_t = \alpha_d r_{d,t} + (1 - \alpha_d) \left[ (1 + r_{f,t})(1 + \Delta tc_t) - 1 \right]
\end{equation}

\section{Propuestas Cuantitativas de Vanguardia para Discutir con el Jurado}

\subsection{Propuesta 1: Proceso de Difusión Estocástica Cox-Ingersoll-Ross (CIR)}

Para evitar calibraciones estáticas de la tasa externa $r_{f,t}$, modelamos el EMBI+ como una ecuación diferencial estocástica afín con reversión a la media:

\begin{equation}
d(risk_t) = \kappa (\theta - risk_t) dt + \sigma \sqrt{risk_t} \, dW_t
\end{equation}

\begin{mdframed}[style=mathbox]
\textbf{Condición de Feller (No Negatividad Estricta):}
\begin{equation}
2\kappa\theta > \sigma^2 \quad \Longleftrightarrow \quad \text{Ratio de Feller} = \frac{2\kappa\theta}{\sigma^2} > 1
\end{equation}
Si el ratio supera la unidad, la frontera en cero es inaccesible con probabilidad 1.
\end{mdframed}

\textbf{Función de Transición Exacta (Chi-Cuadrado no central)}:

\begin{equation}
f(risk_{t+\Delta t} \mid risk_t) = c \, e^{-u - v} \left(\frac{v}{u}\right)^{q/2} I_q\left(2\sqrt{uv}\right)
\end{equation}

Donde $c = \frac{2\kappa}{\sigma^2 (1 - e^{-\kappa \Delta t})}$, $u = c \cdot risk_t e^{-\kappa \Delta t}$, $v = c \cdot risk_{t+\Delta t}$, $q = \frac{2\kappa\theta}{\sigma^2} - 1$, e $I_q(\cdot)$ es la función modificada de Bessel de primera especie.

\textbf{Extensión a 2 Factores (Corto y Largo Plazo)}:
\begin{align}
risk_t &= \xi_t + \chi_t \\
d\xi_t &= -\kappa_\xi \xi_t dt + \sigma_\xi \sqrt{\xi_t} \, dW_t^{(1)} \quad (\text{Shock de liquidez global transitorio, } \kappa_\xi > 1.5) \\
d\chi_t &= \kappa_\chi (\theta_\chi - \chi_t) dt + \sigma_\chi \sqrt{\chi_t} \, dW_t^{(2)} \quad (\text{Solvencia estructural local, } \kappa_\chi \approx 0.2)
\end{align}

\subsection{Propuesta 2: Difusión con Saltos de Merton y Variación Bipotencia}

Para modelar los eventos de corte súbito de crédito (*sudden stops*) y crisis de balanza de pagos:

\begin{equation}
\frac{d(risk_t)}{risk_t} = (\mu - \lambda \kappa_J) dt + \sigma dW_t + (J_t - 1) dN_t
\end{equation}

Donde $N_t \sim \text{Poisson}(\lambda)$ y $\ln(J_t) \sim \mathcal{N}(\mu_J, \sigma_J^2)$. La separación de la volatilidad continua de los saltos discretos se realiza mediante la **Variación Bipotencia de Barndorff-Nielsen y Shephard (2004)**:

\begin{equation}
BV = \frac{\pi}{2} \sum_{t=2}^{T} |r_t| |r_{t-1}| \xrightarrow{p} \int_0^T \sigma_s^2 ds
\end{equation}

\subsection{Propuesta 3: Threshold VECM Multivariado (Hansen y Seo, 2002)}

\begin{equation}
\Delta Y_t = \begin{cases}
\mu_1 + \alpha_1 \beta^\top Y_{t-1} + \sum_{i=1}^{p-1} \Gamma_{1,i} \Delta Y_{t-i} + \varepsilon_t & \text{si } risk_{t-1} \le \tau^* \\
\mu_2 + \alpha_2 \beta^\top Y_{t-1} + \sum_{i=1}^{p-1} \Gamma_{2,i} \Delta Y_{t-i} + \varepsilon_t & \text{si } risk_{t-1} > \tau^*
\end{cases}
\end{equation}

\section{Guion de Defensa Estratégica ante el Profesor}

\begin{mdframed}[style=infobox]
\textbf{1. Pregunta sobre el coeficiente nulo ($\rho = -0.0071$, $\alpha_{pb} = 0.0044$):}\\
\textit{Respuesta}: No es un fallo muestral ni econométrico, sino el hallazgo central: Argentina careció de una regla fiscal activa en el período post-convertibilidad. El ajuste de solvencia no operó por esfuerzo primario presupuestario, sino por canales pasivos de licuación (devaluaciones del TCRM, inflación) y quitas de deuda en reestructuraciones soberanas.

\vspace{0.2cm}
\textbf{2. Pregunta sobre la sensibilidad del test de Johansen al año 1999:}\\
\textit{Respuesta}: Los quiebres extremos de 2001--2002 reducen la potencia del test de Traza (Johansen et al., 2000). Por eso reportamos la sensibilidad con transparencia e imponemos $r=1$ por fundamento teórico de Bohn (1998), demostrando que en la submuestra 2004--2025 Johansen confirma $r=1$ de forma endógena con el mismo resultado nulo.

\vspace{0.2cm}
\textbf{3. Pregunta sobre el test de Hansen en niveles vs diferencias:}\\
\textit{Respuesta}: Dado que $pb_t$ es $I(1)$, la estimación en niveles viola el supuesto de estacionariedad estricta de Hansen (2000). Por eso se reestimó en $\Delta pb_t$ (estacionaria), donde el test por remuestreo Sup-LM arroja $p < 0.001$ y confirma fatiga fiscal nítida a partir de 2083 pb.
\end{mdframed}
